ID_PAD_RULES = {"obligorNumber": 11, "obligationNumber": 10}

def clean_and_pad(record):
    out = dict(record)
    for field, width in ID_PAD_RULES.items():
        if field in out and out[field]:
            out[field] = str(out[field]).zfill(width)
    return out

def _load_sample_data():
    # Only ever called from inside local_mode (see run() below) -- a real
    # DataflowRunner run never touches this file, so it no longer needs
    # to exist on the machine launching a real job.
    sample_data = json.load(open("dummy_all11_with_topicname.json"))
    return [r for recs in sample_data.values() for r in recs]

def row_key_for(record, pk_field, fk_fields):
    if fk_fields:
        return f"{record[fk_fields[0]]}#{record[pk_field]}"
    return record[pk_field]

def to_bigtable_mutation(record, pk_field, fk_fields):
    from google.cloud.bigtable.row import DirectRow
    row = DirectRow(row_key=row_key_for(record, pk_field, fk_fields).encode("utf-8"))
    for k, v in record.items():
        row.set_cell("data", k, str(v) if v is not None else "", timestamp=None)
    return row

class WriteToAlloyDB(beam.DoFn):
    def __init__(self, connection_name, user, password, db, sql_table, columns):
        self.connection_name, self.user, self.password, self.db = connection_name, user, password, db
        self.sql_table, self.columns = sql_table, columns
        self.connector = self.conn = None

    def setup(self):
        from google.cloud.alloydb.connector import Connector
        self.connector = Connector()
        self.conn = self.connector.connect(self.connection_name, "pg8000",
                                            user=self.user, password=self.password, db=self.db)

    def process(self, record):
        cols = ", ".join(self.columns)
        placeholders = ", ".join(["%s"] * len(self.columns))
        values = [record.get(c) for c in self.columns]
        sql = f"INSERT INTO {self.sql_table} ({cols}) VALUES ({placeholders})"
        cur = self.conn.cursor(); cur.execute(sql, values); self.conn.commit(); cur.close()
        yield record

    def teardown(self):
        if self.conn: self.conn.close()
        if self.connector: self.connector.close()

def run():
    parser = argparse.ArgumentParser()
    parser.add_argument("--project_for_subs", default=None)
    parser.add_argument("--bigtable_instance", default=None)
    parser.add_argument("--alloydb_connection", default=None)
    parser.add_argument("--alloydb_user", default=None)
    parser.add_argument("--alloydb_password", default=None)
    parser.add_argument("--alloydb_db", default=None)
    known_args, pipeline_args = parser.parse_known_args()
    local_mode = known_args.project_for_subs is None
    options = PipelineOptions(pipeline_args, streaming=not local_mode)

    with beam.Pipeline(options=options) as p:
        if local_mode:
            all_sample_messages = _load_sample_data()
            loan_msgs = p | "Sample loan" >> beam.Create(
                [r for r in all_sample_messages if r["topic_name"] != "customer-supplementals"])
            cust_msgs = p | "Sample cust" >> beam.Create(
                [r for r in all_sample_messages if r["topic_name"] == "customer-supplementals"])
        else:
            loan_sub = f"projects/{known_args.project_for_subs}/subscriptions/common-loan-topic-sub"
            cust_sub = f"projects/{known_args.project_for_subs}/subscriptions/common-customer-topic-sub"
            loan_msgs = (p | "Read loan" >> beam.io.ReadFromPubSub(subscription=loan_sub)
                           | "Parse loan" >> beam.Map(lambda b: json.loads(b.decode("utf-8"))))
            cust_msgs = (p | "Read cust" >> beam.io.ReadFromPubSub(subscription=cust_sub)
                           | "Parse cust" >> beam.Map(lambda b: json.loads(b.decode("utf-8"))))

        for code, topic_name_val, real_topic_val, pk_field, fk_fields, bt_table, columns in BRANCHES:
            source = cust_msgs if real_topic_val == "common-customer-topic" else loan_msgs
            records = source | f"Filter {code}" >> beam.Filter(
                lambda r, tn=topic_name_val: r.get("topic_name") == tn)
            cleaned = records | f"Clean {code}" >> beam.Map(clean_and_pad)

            if local_mode or not known_args.bigtable_instance:
                cleaned | f"PrintBT {code}" >> beam.Map(
                    lambda r, pk=pk_field, fk=fk_fields, bt=bt_table: print(f"[Bigtable:{bt}] key={row_key_for(r,pk,fk)}"))
            else:
                from apache_beam.io.gcp.bigtableio import WriteToBigTable
                project_id = options.view_as(GoogleCloudOptions).project
                (cleaned | f"BuildBT {code}" >> beam.Map(to_bigtable_mutation, pk_field=pk_field, fk_fields=fk_fields)
                         | f"WriteBT {code}" >> WriteToBigTable(project_id=project_id,
                               instance_id=known_args.bigtable_instance, table_id=bt_table))

            if local_mode or not known_args.alloydb_connection:
                cleaned | f"PrintSQL {code}" >> beam.Map(
                    lambda r, t=bt_table, c=columns: print(
                        f"[AlloyDB:{t}] INSERT ({chr(44).join(c)}) VALUES ({chr(44).join(repr(r.get(x)) for x in c)})"))
            else:
                cleaned | f"WriteSQL {code}" >> beam.ParDo(WriteToAlloyDB(
                    connection_name=known_args.alloydb_connection, user=known_args.alloydb_user,
                    password=known_args.alloydb_password, db=known_args.alloydb_db,
                    sql_table=bt_table, columns=columns))

if __name__ == "__main__":
    run()
